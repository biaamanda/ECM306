public Componente buscar(int codigo) {

        int indiceInicial = hash(codigo);

        for (int i = 0; i < TAMANHO; i++) {
            int indice = (indiceInicial + i) % TAMANHO;

            if (tabela[indice] == null) {
                return null;
            }
            if (tabela[indice].getCodigo() == codigo) {
                return tabela[indice];
            }
        }

        return null;
    }
